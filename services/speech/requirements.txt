# SAMVAAD speech service — base dependencies
#
# Deliberately light: the service must boot in seconds on a free-tier host and
# report honestly which analysis stages are available. The heavy ML stack lives
# in requirements-ml.txt and is optional.

fastapi>=0.115,<0.116
uvicorn[standard]>=0.32,<0.33
pydantic>=2.9,<3.0
pydantic-settings>=2.6,<3.0

# Structured logging, request tracing, redaction, service auth and rate limiting,
# shared with the other two services. Each of these only works if all three
# obey the same version of it — see packages/platform/README.md.
-e ../../packages/platform

# audio handling and the GOP maths
numpy>=2.0,<3.0
soundfile>=0.12,<1.0

# grapheme-to-phoneme: CMUdict lookup with a neural fallback for out-of-vocabulary
# words. Small, and it needs one NLTK download, which `warm_g2p` performs:
#   python -m scripts.warm_g2p
g2p-en>=2.1,<3.0

# Prosody and disfluency FEATURES — not the ML stack.
#
# These sit in the base tier rather than in requirements-ml.txt because prosody
# (M7 part A) is deterministic signal processing, not a model: speech rate,
# pause structure, F0 and energy need no weights and no GPU. Keeping them here
# means a free-tier CPU host can serve real fluency analytics with no torch
# installed at all, and it means the train/serve parity tests over
# `extract_features` actually run in CI instead of being skipped.
#
# Together they add roughly 40 MB, against 2.5 GB for the torch tier.
librosa>=0.10,<1.0
praat-parselmouth>=0.4,<1.0
